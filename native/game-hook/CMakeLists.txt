cmake_minimum_required(VERSION 3.16)
project(NolaGameHook LANGUAGES CXX)
add_executable(nola-game-hook main.cpp)
target_compile_features(nola-game-hook PRIVATE cxx_std_17)
target_compile_options(nola-game-hook PRIVATE /utf-8)
set_property(TARGET nola-game-hook PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_executable(nola-game-hook-test-target test-target.cpp)
target_compile_options(nola-game-hook-test-target PRIVATE /utf-8)
set_property(TARGET nola-game-hook-test-target PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
